\pagenumbering{Roman}

\thispagestyle{plain} %Entfernt Kopfzeile (bzw. setzt die Seite auf LaTeX-Standard zurück)
\tableofcontents %Fügt Inhaltsverzeichnis ein, jede Sektion wird automatisch hinzugefügt.

\newpage
\thispagestyle{plain}
\listoffigures %Fügt Abbildungsverzeichnis ein

\clearpage %Benötigt, um fortan mit Zahlen zu nummerieren, sonst würde der Befehl in der nächsten Zeile den vorherigen Numerierungsstyle überschreiben

\input{Inhaltliches/Einleitung} %Der Übersichthalber wird die Einleitung in einer extra Datei geschrieben

\input{Inhaltliches/Zusammenfassung} %Der Übersichthalber wird die Zusammenfassung in einer extra Datei geschrieben

\newpage
\input{Inhaltliches/Literatur} %Der Übersichthalber wird die Literaturangabe in einer extra Datei geschrieben

\newpage
\input{Inhaltliches/Anhang} %Der Übersichthalber wird der Anhang in einer extra Datei geschrieben

\printbibliography %Literaturverzeichnis aus literatur.bib
