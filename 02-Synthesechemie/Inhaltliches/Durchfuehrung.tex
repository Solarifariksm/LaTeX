\section{Durchführung}
\label{sec:durchfuehrung}

\subsection{Apparatur}
Reaktionsaufbau: 100-mL-Rundkolben mit Rückflusskühler, Magnetrührer,
Trockenrohr (CaCl\textsubscript{2}).

  \centering

  \caption{Platzhalter: Skizze/Foto der Versuchsapparatur.}
  \label{fig:apparatur}
\end{figure}

\subsection{Reaktionsdurchführung}
\begin{enumerate}
  \item 2-Methyl-2-buten (\SI{10.0}{\milli\liter}) mit Wasser und
        Katalysator in den Rundkolben geben.
  \item Unter Rühren 30 min zum Rückfluss erhitzen.
  \item Nach Abkühlen auf RT die organische Phase abtrennen.
\end{enumerate}

\subsection{Aufreinigung}
Destillation bei Normaldruck (Vigreux-Kolonne), Vorlauf verworfen.

\subsection{Sicherheit}
\begin{itemize}
  \item \ce{H2SO4} konz.: schwere Verätzungen – Handschuhe, Schutzbrille!
  \item 2-Methyl-2-buten: leichtentzündlich ($F$), von Zündquellen fernhalten.
  \item Abfall: organische Phasen in Behälter „halogenfreie organische Abfälle“.
\end{itemize}
